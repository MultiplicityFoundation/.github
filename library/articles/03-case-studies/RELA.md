def \_\_init\_\_(self, n\_phases=5, n\_actions=2):

shift\]

self.alpha = 0.1 \# Learning rate

self.epsilon = 0.2 \# Exploration rate

def choose\_action(self, phase):

if np.random.uniform() \< self.epsilon:

return np.random.choice(\[0, 1\]) \# Random explore

return np.argmax(self.q\_table\[phase\])

def update\_q\_table(self, phase, action, reward, next\_phase):

best\_next = np.max(self.q\_table\[next\_phase\])

self.q\_table\[phase, action\] += self.alpha \* (

reward + self.gamma \* best\_next - self.q\_table\[phase, action\]

)
