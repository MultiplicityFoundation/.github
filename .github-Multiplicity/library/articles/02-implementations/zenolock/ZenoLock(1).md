from scipy.linalg import svd

from sympy import mersenne\_prime

\# Optimized PIRTM Key Generation

def generate\_pirtm\_key(n\_primes, tensor\_dim):

primes = \[mersenne\_prime(i) for i in range(1, n\_primes + 1)\] \#
Mersenne primes

psi = np.random.random((tensor\_dim, tensor\_dim)) \# Simulated quantum
states

tensor = np.tensordot(psi, psi, axes=0)

U, S, Vt = svd(tensor, full\_matrices=False)

k = min(tensor\_dim, 10) \# Truncated SVD

compressed\_tensor = np.dot(U\[:, :k\] \* S\[:k\], Vt\[:k, :\])

lambda\_coeffs = np.random.random(n\_primes) \# Multiplicity weights

key = sum(lambda\_coeffs\[i\] \* compressed\_tensor \* primes\[i\] for i
in range(n\_primes))

return key

\# Homomorphic Encryption (CKKS-inspired)

def homomorphic\_encrypt(message, key):

s = 0.5 + 1j \* np.random.random() \# Langlands L-function parameter

l\_value = np.sum(\[1 / (n \*\* s) for n in range(1, 100)\]) \#
Approximate L-function

enc\_message = message \* l\_value \* key

def homomorphic\_add(enc\_m1, enc\_m2):

return enc\_m1 + enc\_m2 \# FHE addition

return enc\_message, homomorphic\_add
